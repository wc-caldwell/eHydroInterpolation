def create_TIN_mask(triangulation_in, max_edge_length_in):
    """
    THIS IS AI GENERATED. Need to debug and make sure this actually does what I expect of it.
    Beginning from a Matplotlib triangulation, mask out any triangles with an edge length exceeding some threshold.
    :param triangulation_in: triangulation using Matplotlib.tri.Triangulation
    :param max_edge_length_in: the DTM number.
    :return:
    """
    # Get the coordinates of the triangle vertices.
    triangles = triangulation_in.triangles
    x_coords = triangulation_in.x[triangles]
    y_coords = triangulation_in.y[triangles]

    # Calculate edge lengths for each triangle.
    edge1 = np.sqrt((x_coords[:, 0] - x_coords[:, 1]) ** 2 +
                    (y_coords[:, 0] - y_coords[:, 1]) ** 2)
    edge2 = np.sqrt((x_coords[:, 1] - x_coords[:, 2]) ** 2 +
                    (y_coords[:, 1] - y_coords[:, 2]) ** 2)
    edge3 = np.sqrt((x_coords[:, 2] - x_coords[:, 0]) ** 2 +
                    (y_coords[:, 2] - y_coords[:, 0]) ** 2)

    # Find the maximum edge length for each triangle.
    max_edge_length = np.maximum(np.maximum(edge1, edge2), edge3)

    # Create the mask
    mask_out = max_edge_length > max_edge_length_in
    return mask_out
